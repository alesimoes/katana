### generic last or default
@ok
@has fn lastOr: (list['a], 'a) -> 'a
fn lastOr(xs, d)
  if len(xs) == 0
    return d
  return xs[len(xs) - 1]
### generic pairs of two lists as a list of lists
@ok
@has fn zip2: (list['a], list['a]) -> list[list['a]]
fn zip2(a, b)
  let out = []
  let n = min(len(a), len(b))
  for i in 0..n
    push(out, [a[i], b[i]])
  return out

### generic grouping into a map of counts
@ok
@has fn tally: (list[str]) -> map[str,int]
fn tally(xs)
  let m = {}
  for x in xs
    if contains(m, x)
      m[x] += 1
    else
      m[x] = 1
  return m
